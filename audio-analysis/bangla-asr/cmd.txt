# Setup Commands for Pyannote Audio Speaker Diarization & Whisper ASR

# 1. Install FFmpeg (required for audio conversion)
conda install -y ffmpeg -c conda-forge

# 2. Install Python dependencies
pip install -r requirements.txt

pip install -U "nemo_toolkit[asr]"

# ============================================================
# USAGE
# ============================================================

# --- Audio Conversion ---

# Convert single file (output in same directory):
python convert.py --input-filepath input.webm --output-format wav

# Convert single file to specific path:
python convert.py --input-filepath input.webm --output-filepath output.wav

# Convert all files in a directory:
python convert.py --input-dir ./videos --output-dir ./audio --output-format wav

# --- Speaker Diarization (pyannote) ---

# Run diarization on a WAV file:
python diarize.py --input audio.wav --output diarization.csv

# --- Speech Recognition (Whisper/Tugstugi) ---

# Run Bengali ASR on a WAV file:
python tugstugi.py --input-filepath audio.wav --output-filepath transcript.txt

# With options:
python tugstugi.py --input-filepath audio.wav --output-filepath transcript.txt --cpu

# ============================================================
# EXAMPLE WORKFLOW
# ============================================================

# Step 1: Convert to wav
python convert.py --input-filepath test_1.mp3 --output-format wav

# Step 2: Run diarization
python diarize.py --input test_1.wav --output test_1_diarization.csv

# Step 3: Run speech recognition
python tugstugi.py --input-filepath test_1.wav --output-filepath test_1_transcript_tugstugi.txt
